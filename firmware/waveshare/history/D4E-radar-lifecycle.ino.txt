    double longitude = 0.0;

    bool directionValid = false;
    float direction = 0.0f;

    bool horizontalSpeedValid = false;
    float horizontalSpeed = 0.0f;

    bool verticalSpeedValid = false;
    float verticalSpeed = 0.0f;

    bool barometricAltitudeValid = false;
    float barometricAltitude = 0.0f;

    bool geodeticAltitudeValid = false;
    float geodeticAltitude = 0.0f;

    bool heightValid = false;

    // Accuracy
    // --------------------------------

    uint8_t verticalAccuracy = 0;
    uint8_t barometricAccuracy = 0;
    uint8_t speedAccuracy = 0;
    uint8_t timestampAccuracy = 0;


    // --------------------------------
    // RID timestamp
    // --------------------------------

    // --------------------------------
    // System message
    // --------------------------------

    bool hasSystemData = false;

    uint8_t operatorLocationType = 0;
    uint8_t classificationType = 0;

    bool hasSystemOperatorPosition = false;

    double systemOperatorLatitude = 0.0;
    double systemOperatorLongitude = 0.0;

    uint16_t areaCount = 0;
    uint16_t areaRadius = 0;

    bool areaCeilingValid = false;

    uint8_t classEU = 0;

    // --------------------------------

// ====================================================
// BYTE HELPERS
// ====================================================

uint8_t getByte(
    const String &data,
    size_t index
) {

    return static_cast<uint8_t>(
        data[index]
    );
}


uint16_t readU16LE(
    const String &data,
    size_t index
) {

    uint16_t value = 0;

    value |=
        static_cast<uint16_t>(
            getByte(data, index)
        );

    value |=
        static_cast<uint16_t>(
            getByte(data, index + 1)
        ) << 8;

    return value;
}


int32_t readI32LE(
    const String &data,
    size_t index
) {

    uint32_t value = 0;

    value |=
        static_cast<uint32_t>(
            getByte(data, index)
        );

    value |=
        static_cast<uint32_t>(
            getByte(data, index + 1)
        ) << 8;

    value |=
        static_cast<uint32_t>(
            getByte(data, index + 2)
        ) << 16;

    value |=
        static_cast<uint32_t>(
            getByte(data, index + 3)
        ) << 24;

    return static_cast<int32_t>(
        value
    );
}


String readTextField(
    const String &data,
    size_t startIndex,
    size_t length
) {

    String text = "";

    for (
        size_t i = 0;
        i < length;
        i++
    ) {

        char c =
            data[startIndex + i];

        if (c == '\0') {
            break;
        }

        if (
            c >= 32
            &&
            c <= 126
        ) {

            text += c;
        }
    }

    return text;
}

// ====================================================
// TARGET LIFECYCLE
// ====================================================

const char *targetStateName(
    const RIDTarget &target
) {

    if (!target.allocated) {
        return "EMPTY";
    }


    uint32_t age =
        millis()
        -
        target.lastSeenMs;


    if (
        age <= TARGET_ACTIVE_MS
    ) {

        return "ACTIVE";
    }


    if (
        age <= TARGET_STALE_MS
    ) {

        return "STALE";
    }


    return "LOST";
}


// ====================================================
// RSSI FILTER
// ====================================================

void addRSSISample(
    RIDTarget &target,
    int rssi
) {

    target.currentRSSI =
        rssi;


    target.rssiSamples[
        target.rssiIndex
    ] = rssi;


    target.rssiIndex++;


    if (
        target.rssiIndex >= 5
    ) {

        target.rssiIndex = 0;
    }


    if (
        target.rssiCount < 5
    ) {

        target.rssiCount++;
    }
}


int getFilteredRSSI(
    RIDTarget &target
) {

    if (
        target.rssiCount == 0
    ) {

        return 0;
    }


    int sorted[5];


    for (
        uint8_t i = 0;
        i < target.rssiCount;
        i++
    ) {

        sorted[i] =
            target.rssiSamples[i];
    }


    for (
        uint8_t i = 1;
        i < target.rssiCount;
        i++
    ) {

        int key =
            sorted[i];

        int j =
            i - 1;


        while (
            j >= 0
            &&
            sorted[j] > key
        ) {

            sorted[j + 1] =
                sorted[j];

            j--;
        }


        sorted[j + 1] =
            key;
    }


    if (
        target.rssiCount % 2 == 1
    ) {

        return
            sorted[
                target.rssiCount / 2
            ];
    }


    int upper =
        sorted[
            target.rssiCount / 2
        ];


    int lower =
        sorted[
            (target.rssiCount / 2) - 1
        ];


    return
        (lower + upper)
        / 2;
}


// ====================================================
// TARGET LOOKUP / ALLOCATION
// ====================================================

int findTargetByAddress(
    const String &address
) {

    for (
        uint8_t i = 0;
        i < MAX_TARGETS;
        i++
    ) {

        if (
            targets[i].allocated
            &&
            targets[i].sourceAddress
                .equalsIgnoreCase(address)
        ) {

            return i;
        }
    }


    return -1;
}


int allocateTarget(
    const String &address
) {

    // --------------------------------
    // First try an unused slot.
    // --------------------------------

    for (
        uint8_t i = 0;
        i < MAX_TARGETS;
        i++
    ) {

        if (
            !targets[i].allocated
        ) {

            targets[i] =
                RIDTarget();


            targets[i].allocated =
                true;


            targets[i].sourceAddress =
                address;


            targets[i].firstSeenMs =
                millis();


            targets[i].lastSeenMs =
                millis();


            return i;
        }
    }


    // --------------------------------
    // Otherwise reuse a sufficiently
    // old LOST target.
    // --------------------------------

    for (
        uint8_t i = 0;
        i < MAX_TARGETS;
        i++
    ) {

        uint32_t age =
            millis()
            -
            targets[i].lastSeenMs;


        if (
            age > TARGET_REUSE_MS
        ) {

            targets[i] =
                RIDTarget();


            targets[i].allocated =
                true;


            targets[i].sourceAddress =
                address;


            targets[i].firstSeenMs =
                millis();


            targets[i].lastSeenMs =
                millis();


            return i;
        }
    }


    targetAllocationFailures++;

    return -1;
}


int findOrCreateTarget(
    const String &address
) {

    int targetIndex =
        findTargetByAddress(
            address
        );


    if (
        targetIndex >= 0
    ) {

        return targetIndex;
    }


    return
        allocateTarget(
            address
        );
}


// ====================================================
// TELEMETRY DECODERS
// ====================================================

float decodeAltitude(
    uint16_t encodedAltitude
) {

    return
        (
            static_cast<float>(
                encodedAltitude
            )
            * 0.5f
        )
        - 1000.0f;
}


float decodeHorizontalSpeed(
    uint8_t encodedSpeed,
    bool speedMultiplier
) {

    if (
        speedMultiplier
    ) {

        return
            (
                static_cast<float>(
                    encodedSpeed
                )
                * 0.75f
            )
            + 63.75f;
    }


    return
        static_cast<float>(
            encodedSpeed
        )
        * 0.25f;
}


float decodeVerticalSpeed(
    int8_t encodedSpeed
) {

    return
        static_cast<float>(
            encodedSpeed
        )
        * 0.5f;
}


// ====================================================
// BASIC ID UPDATE
// ====================================================

void updateBasicID(
    RIDTarget &target,
    const String &serviceData
) {

    target.basicIDCount++;


    uint8_t idAndUAType =
        getByte(
            serviceData,
            3
        );


    uint8_t idType =
        (
            idAndUAType >> 4
        )
        & 0x0F;


    uint8_t uaType =
        idAndUAType
        & 0x0F;


    String uasID =
        readTextField(
            serviceData,
            4,
            20
        );


    target.uaType =
        uaType;


    if (
        idType == 1
        &&
        uasID.length() > 0
    ) {

        target.serialID =
            uasID;

        target.hasSerialID =
            true;
    }


    else if (
        idType == 2
        &&
        uasID.length() > 0
    ) {

        target.registrationID =
            uasID;

        target.hasRegistrationID =
            true;
    }
}


// ====================================================
// LOCATION UPDATE
// ====================================================

void updateLocation(
    RIDTarget &target,
    const String &serviceData
) {

    target.locationCount++;


    uint8_t statusFlags =
        getByte(
            serviceData,
            3
        );


    bool speedMultiplier =
        statusFlags
        & 0x01;


    bool eastWestDirection =
        (
            statusFlags >> 1
        )
        & 0x01;


    bool heightType =
        (
            statusFlags >> 2
        )
        & 0x01;


    target.flightStatus =
        (
            statusFlags >> 4
        )
        & 0x0F;


    uint8_t directionEncoded =
        getByte(
            serviceData,
            4
        );


    float direction =
        static_cast<float>(
            directionEncoded
        );


    if (
        eastWestDirection
    ) {

        direction += 180.0f;
    }


    if (
        direction <= 360.0f
    ) {

        target.direction =
            direction;

        target.directionValid =
            true;
    }

    else {

        target.directionValid =
            false;
    }


    uint8_t horizontalSpeedEncoded =
        getByte(
            serviceData,
            5
        );


    float horizontalSpeed =
        decodeHorizontalSpeed(
            horizontalSpeedEncoded,
            speedMultiplier
        );


    if (
        horizontalSpeed < 255.0f
    ) {

        target.horizontalSpeed =
            horizontalSpeed;

        target.horizontalSpeedValid =
            true;
    }

    else {

        target.horizontalSpeedValid =
            false;
    }


    int8_t verticalSpeedEncoded =
        static_cast<int8_t>(
            getByte(
                serviceData,
                6
            )
        );


    float verticalSpeed =
        decodeVerticalSpeed(
            verticalSpeedEncoded
        );


    if (
        verticalSpeed != 63.0f
    ) {

        target.verticalSpeed =
            verticalSpeed;

        target.verticalSpeedValid =
            true;
    }

    else {

        target.verticalSpeedValid =
            false;
    }


    int32_t latitudeRaw =
        readI32LE(
            serviceData,
            7
        );


    int32_t longitudeRaw =
        readI32LE(
            serviceData,
            11
        );


    target.latitude =
        static_cast<double>(
            latitudeRaw
        )
        / 10000000.0;


    target.longitude =
        static_cast<double>(
            longitudeRaw
        )
        / 10000000.0;


    target.hasPosition =
        (
            latitudeRaw != 0
            ||
            longitudeRaw != 0
        );


    if (target.hasPosition) {

        target.lastPositionSeenMs =
            millis();
    }


    uint16_t baroRaw =
        readU16LE(
            serviceData,
            15
        );


    uint16_t geoRaw =
        readU16LE(
            serviceData,
            17
        );


    uint16_t heightRaw =
        readU16LE(
            serviceData,
            19
        );


    target.barometricAltitudeValid =
        baroRaw != 0;


    target.geodeticAltitudeValid =
        geoRaw != 0;


    target.heightValid =
        heightRaw != 0;


    if (
        target.barometricAltitudeValid
    ) {

        target.barometricAltitude =
            decodeAltitude(
                baroRaw
            );
    }


    if (
        target.geodeticAltitudeValid
    ) {

        target.geodeticAltitude =
            decodeAltitude(
                geoRaw
            );
    }


    if (
        target.heightValid
    ) {

        target.height =
            decodeAltitude(
                heightRaw
            );
    }


    target.heightFromGround =
        heightType;


    uint8_t accuracyByte1 =
        getByte(
            serviceData,
            21
        );


    target.horizontalAccuracy =
        accuracyByte1
        & 0x0F;


    target.verticalAccuracy =
        (
            accuracyByte1 >> 4
        )
        & 0x0F;


    uint8_t accuracyByte2 =
        getByte(
            serviceData,
            22
        );


    target.speedAccuracy =
        accuracyByte2
        & 0x0F;


    target.barometricAccuracy =
        (
            accuracyByte2 >> 4
        )
        & 0x0F;


    uint16_t timestampRaw =
        readU16LE(
            serviceData,
            23
        );


    if (
        timestampRaw
        !=
        0xFFFF
    ) {

        target.remoteIDTimestamp =
            static_cast<float>(
                timestampRaw
            )
            / 10.0f;
    }


    target.timestampAccuracy =
        getByte(
            serviceData,
            25
        )
        & 0x0F;
}


// ====================================================
// SYSTEM UPDATE
// ====================================================

void updateSystem(
    RIDTarget &target,
    const String &serviceData
) {

    target.systemCount++;

    target.hasSystemData =
        true;


    // --------------------------------
    // Operator location /
    // classification flags
    // --------------------------------

    uint8_t flags =
        getByte(
            serviceData,
            3
        );


    target.operatorLocationType =
        flags
        &
        0x03;


    target.classificationType =
        (
            flags >> 2
        )
        &
        0x07;


    // --------------------------------
    // Operator position
    // --------------------------------

    int32_t operatorLatitudeRaw =
        readI32LE(
            serviceData,
            4
        );


    int32_t operatorLongitudeRaw =
        readI32LE(
            serviceData,
            8
        );


    if (
        operatorLatitudeRaw != 0
        ||
        operatorLongitudeRaw != 0
    ) {

        target.systemOperatorLatitude =
            static_cast<double>(
                operatorLatitudeRaw
            )
            /
            10000000.0;


        target.systemOperatorLongitude =
            static_cast<double>(
                operatorLongitudeRaw
            )
            /
            10000000.0;


        target.hasSystemOperatorPosition =
            true;
    }

    else {

        target.systemOperatorLatitude =
            0.0;

        target.systemOperatorLongitude =
            0.0;

        target.hasSystemOperatorPosition =
            false;
    }


    // --------------------------------
    // Area data
    // --------------------------------

    target.areaCount =
        readU16LE(
            serviceData,
            12
        );


    target.areaRadius =
        static_cast<uint16_t>(
            getByte(
                serviceData,
                14
            )
        )
        *
        10U;


    // --------------------------------
    // Area ceiling
    // --------------------------------

    uint16_t areaCeilingRaw =
        readU16LE(
            serviceData,
            15
        );


    if (
        areaCeilingRaw
        !=
        0
    ) {

        target.areaCeiling =
            decodeAltitude(
                areaCeilingRaw
            );

        target.areaCeilingValid =
            true;
    }

    else {

        target.areaCeilingValid =
            false;
    }


    // --------------------------------
    // Area floor
    // --------------------------------

    uint16_t areaFloorRaw =
        readU16LE(
            serviceData,
            17
        );


    if (
        areaFloorRaw
        !=
        0
    ) {

        target.areaFloor =
            decodeAltitude(
                areaFloorRaw
            );

        target.areaFloorValid =
            true;
    }

    else {

        target.areaFloorValid =
            false;
    }


    // --------------------------------
    // EU category / class
    // --------------------------------

    uint8_t classificationByte =
        getByte(
            serviceData,
            19
        );


    target.classEU =
        classificationByte
        &
        0x0F;


    target.categoryEU =
        (
            classificationByte >> 4
        )
        &
        0x0F;


    // --------------------------------
    // Operator geodetic altitude
    // --------------------------------

    uint16_t operatorAltitudeRaw =
        readU16LE(
            serviceData,
            20
        );


    if (
        operatorAltitudeRaw
        !=
        0
    ) {

        target.operatorGeoAltitude =
            decodeAltitude(
                operatorAltitudeRaw
            );

        target.operatorGeoAltitudeValid =
            true;
    }

    else {

        target.operatorGeoAltitudeValid =
            false;
    }


    // --------------------------------
    // System timestamp
    //
    // uint32 little-endian,
    // seconds since 2019-01-01 UTC.
    // --------------------------------

    target.systemTimestamp =
        static_cast<uint32_t>(
            getByte(
                serviceData,
                22
            )
        )
        |
        (
            static_cast<uint32_t>(
                getByte(
                    serviceData,
                    23
                )
            )
            << 8
        )
        |
        (
            static_cast<uint32_t>(
                getByte(
                    serviceData,
                    24
                )
            )
            << 16
        )
        |
        (
            static_cast<uint32_t>(
                getByte(
                    serviceData,
                    25
                )
            )
            << 24
        );
}


// ====================================================
// OPERATOR ID UPDATE
// ====================================================

void updateOperatorID(
    RIDTarget &target,
    const String &serviceData
) {

    target.operatorIDCount++;


    String operatorID =
        readTextField(
            serviceData,
            4,
            20
        );


    if (
        operatorID.length() > 0

            operatorID;

        target.hasOperatorID =
            true;
    }

    else {

        target.operatorID =
            "";

        target.hasOperatorID =
            false;
    }
}

uint8_t getActiveTargetCount() {

    uint8_t count = 0;


    for (
        uint8_t i = 0;
        i < MAX_TARGETS;
        i++
    ) {

        if (
            !targets[i].allocated
        ) {

            continue;
        }

            millis()
            -
            targets[i].lastSeenMs;

            age <= TARGET_ACTIVE_MS
        ) {

            count++;
        }
    }


    return count;
}


// ====================================================
// COMPACT MULTI-TARGET SUMMARY
// ====================================================

void printTargetSummary() {

    // --------------------------------------------------------
    // Copy only the compact dashboard fields while holding the
    // target mutex. Serial output happens AFTER releasing it.
    // --------------------------------------------------------

    struct DashboardTargetCopy {

        bool allocated = false;

        uint8_t stateCode = 0;

        uint8_t uaType = 0;
        uint8_t flightStatus = 0;

        bool hasPosition = false;

        int filteredRSSI = 0;

        uint32_t packetCount = 0;
        uint32_t lastSeenMs = 0;
    };


    DashboardTargetCopy copies[MAX_TARGETS];

    uint8_t trackedCount = 0;
    uint8_t activeCount = 0;

    uint32_t snapshotMs =
        millis();

    OutputMode summaryMode =
        outputMode;


    if (targetMutex != nullptr) {

        xSemaphoreTake(
            targetMutex,
            portMAX_DELAY
        );
    }


    for (
        uint8_t i = 0;
        i < MAX_TARGETS;
        i++
    ) {

        RIDTarget &target =
            targets[i];

        DashboardTargetCopy &copy =
            copies[i];


        copy.allocated =
            target.allocated;


        if (!target.allocated) {

            continue;
        }


        trackedCount++;


        uint32_t age =
            snapshotMs
            -
            target.lastSeenMs;


        if (
            age <= TARGET_ACTIVE_MS
        ) {

            copy.stateCode = 1;
            activeCount++;
        }

        else if (
            age <= TARGET_STALE_MS
        ) {

            copy.stateCode = 2;
        }

        else {

            copy.stateCode = 3;
        }


        copy.uaType =
            target.uaType;

        copy.flightStatus =
            target.flightStatus;

        copy.hasPosition =
            target.hasPosition;

        copy.filteredRSSI =
            getFilteredRSSI(
                target
            );

        copy.packetCount =
            target.packetCount;

        copy.lastSeenMs =
            target.lastSeenMs;
    }


    if (targetMutex != nullptr) {

        xSemaphoreGive(
            targetMutex
        );
    }


    // --------------------------------------------------------
    // Serial work is intentionally outside targetMutex.
    // --------------------------------------------------------

    Serial.println();

    Serial.println(
        "========== RID TARGET SUMMARY =========="
    );


    Serial.print(
        "Output Mode: "
    );

    Serial.println(
        summaryMode == MODE_PRIVATE
        ? "PRIVATE"
        : "OPEN"
    );


    Serial.print(
        "Targets Tracked: "
    );

    Serial.println(
        trackedCount
    );


    Serial.print(
        "Targets Active: "
    );

    Serial.println(
        activeCount
    );


    Serial.println();


    for (
        uint8_t i = 0;
        i < MAX_TARGETS;
        i++
    ) {

        DashboardTargetCopy &copy =
            copies[i];


        if (!copy.allocated) {

            continue;
        }


        Serial.print(
            "T"
        );


        if (
            i + 1 < 10
        ) {

            Serial.print(
                "0"
            );
        }


        Serial.print(
            i + 1
        );


        Serial.print(
            "  "
        );


        switch (
            copy.stateCode
        ) {

            case 1:
                Serial.print(
                    "ACTIVE"
                );
                break;

            case 2:
                Serial.print(
                    "STALE"
                );
                break;

            case 3:
                Serial.print(
                    "LOST"
                );
                break;

            default:
                Serial.print(
                    "EMPTY"
                );
                break;
        }


        Serial.print(
            "  "
        );


        Serial.println(
            uaTypeName(
                copy.uaType
            )
        );


        Serial.print(
            "State: "
        );

        Serial.println(
            flightStatusName(
                copy.flightStatus
            )
        );


        Serial.print(
            "Position: "
        );

        Serial.println(
            copy.hasPosition
            ? "VALID"
            : "UNKNOWN"
        );


        Serial.print(
            "RSSI: "
        );

        Serial.print(
            copy.filteredRSSI
        );

        Serial.println(
            " dBm"
        );


        Serial.print(
            "Packets: "
        );

        Serial.println(
            copy.packetCount
        );


        Serial.print(
            "Last Seen: "
        );

        Serial.print(
            snapshotMs
            -
            copy.lastSeenMs
        );

        Serial.println(
            " ms ago"
        );


        Serial.println();
    }


    Serial.println(
        "========================================"
    );
}


// ============================================================
// MAIN-LOOP DASHBOARD SERVICE
//
// Heavy formatted Serial output must never run inside the BLE
// advertisement callback. RF processing owns the callback;
// diagnostics are serviced independently by the main loop.
// ============================================================

void serviceTargetDashboard() {

    if (
        !scanSessionActive
        || scanStopRequested
    ) {

        return;
    }


    uint32_t now =
        millis();


    if (
        now
        -
        lastDashboardPrintMs
        <
        DASHBOARD_INTERVAL_MS
    ) {

        return;
    }


    // Advance the timer before printing so diagnostic timing
    // cannot cause an immediate duplicate emission.
    lastDashboardPrintMs =
        now;


    printTargetSummary();
}

// ====================================================
// SERIAL COMMANDS
// ====================================================

// D4E-A:
// Explicit declarations because the receiver-reference provider
// is implemented later in this .ino.

bool wsSetManualFixedReceiverReference(
    double latitude,
    double longitude
);

void wsClearReceiverReference();

uint8_t wsReceiverReferenceStatusCode();


void processSerialCommands() {

    static char commandBuffer[64];
    static size_t commandLength = 0;
    static bool discardUntilEOL = false;


    while (
        Serial.available() > 0
    ) {

        int incoming =
            Serial.read();


        if (
            incoming < 0
        ) {
            continue;
        }


        char c =
            (char)incoming;


        if (
            c == '\r'
            ||
            c == '\n'
        ) {

            if (
                discardUntilEOL
            ) {

                discardUntilEOL = false;
                commandLength = 0;
                continue;
            }


            if (
                commandLength == 0
            ) {
                continue;
            }


            commandBuffer[
                commandLength
            ] = '\0';


            if (
                commandLength == 1
            ) {

                char command =
                    commandBuffer[0];


                if (
                    command == 'p'
                    ||
                    command == 'P'
                ) {

                    outputMode =
                        MODE_PRIVATE;


                    Serial.println();

                    Serial.println(
                        "*** OUTPUT MODE -> PRIVATE ***"
                    );


                    Serial.println(
                        "Identifiers and coordinates redacted."
                    );
                }


                else if (
                    command == 'o'
                    ||
                    command == 'O'
                ) {

                    outputMode =
                        MODE_OPEN;


                    Serial.println();

                    Serial.println(
                        "*** OUTPUT MODE -> OPEN ***"
                    );


                    Serial.println(
                        "Full identifiers and coordinates enabled."
                    );
                }


                else if (
                    command == 's'
                    ||
                    command == 'S'
                ) {

                    Serial.println();

                    Serial.println(
                        "*** MULTI-TARGET SNAPSHOT ***"
                    );


                    if (targetMutex != nullptr) {
                        xSemaphoreTake(
                            targetMutex,
                            portMAX_DELAY
                        );
                    }

                    printAllTargetDetails();

                    if (targetMutex != nullptr) {
                        xSemaphoreGive(
                            targetMutex
                        );
                    }
                }
            }


            // ------------------------------------------------
            // D4E-A RECEIVER REFERENCE COMMANDS
            //
            // R,<lat>,<lon>
            // R,STATUS
            // R,CLEAR
            //
            // Coordinates are intentionally never printed back.
            // ------------------------------------------------

            else if (
                commandLength >= 3
                &&
                (
                    commandBuffer[0] == 'R'
                    ||
                    commandBuffer[0] == 'r'
                )
                &&
                commandBuffer[1] == ','
            ) {

                const char *payload =
                    commandBuffer + 2;


                // --------------------------------------------
                // R,CLEAR
                // --------------------------------------------

                if (
                    strcmp(
                        payload,
                        "CLEAR"
                    )
                    == 0
                ) {

                    wsClearReceiverReference();

                    Serial.println();
                    Serial.println(
                        "*** RECEIVER REFERENCE -> NONE ***"
                    );

                    Serial.println(
                        "Manual fixed reference cleared."
                    );
                }


                // --------------------------------------------
                // R,STATUS
                //
                // Deliberately reports state only.
                // Never prints coordinates.
                // --------------------------------------------

                else if (
                    strcmp(
                        payload,
                        "STATUS"
                    )
                    == 0
                ) {

                    Serial.println();

                    const uint8_t referenceStatus =
                        wsReceiverReferenceStatusCode();


                    if (referenceStatus == 1) {

                        Serial.println(
                            "*** RECEIVER REFERENCE: FIXED / VALID ***"
                        );
                    }

                    else if (referenceStatus == 2) {

                        Serial.println(
                            "*** RECEIVER REFERENCE: GNSS / VALID ***"
                        );
                    }

                    else {

                        Serial.println(
                            "*** RECEIVER REFERENCE: NONE ***"
                        );
                    }
                }


                // --------------------------------------------
                // R,<latitude>,<longitude>
                // --------------------------------------------

                else {

                    char *latitudeEnd =
                        nullptr;

                    char *longitudeEnd =
                        nullptr;


                    double latitude =
                        strtod(
                            payload,
                            &latitudeEnd
                        );


                    bool formatValid =
                        (
                            latitudeEnd
                            != payload
                            &&
                            latitudeEnd
                            != nullptr
                            &&
                            *latitudeEnd
                                == ','
                        );


                    double longitude =
                        0.0;


                    if (formatValid) {

                        char *longitudeStart =
                            latitudeEnd + 1;


                        longitude =
                            strtod(
                                longitudeStart,
                                &longitudeEnd
                            );


                        formatValid =
                            (
                                longitudeEnd
                                != longitudeStart
                                &&
                                longitudeEnd
                                != nullptr
                                &&
                                *longitudeEnd
                                    == '\0'
                            );
                    }


                    bool coordinateRangeValid =
                        (
                            formatValid
                            &&
                            isfinite(latitude)
                            &&
                            isfinite(longitude)
                            &&
                            latitude >= -90.0
                            &&
                            latitude <= 90.0
                            &&
                            longitude >= -180.0
                            &&
                            longitude <= 180.0
                        );


                    if (
                        coordinateRangeValid
                        &&
                        wsSetManualFixedReceiverReference(
                            latitude,
                            longitude
                        )
                    ) {

                        Serial.println();

                        Serial.println(
                            "*** RECEIVER REFERENCE -> FIXED / VALID ***"
                        );

                        Serial.println(
                            "Coordinates accepted locally; values not echoed."
                        );
                    }

                    else {

                        Serial.println();

                        Serial.println(
                            "*** RECEIVER REFERENCE REJECTED ***"
                        );

                        Serial.println(
                            "Use: R,<latitude>,<longitude>"
                        );

                        Serial.println(
                            "Existing valid reference was not intentionally replaced."
                        );
                    }
                }
            }


            commandLength = 0;
            continue;
        }


        if (
            discardUntilEOL
        ) {
            continue;
        }


        if (
            commandLength <
            sizeof(commandBuffer) - 1
        ) {

            commandBuffer[
                commandLength++
            ] = c;
        }

        else {

            commandLength = 0;
            discardUntilEOL = true;
        }
    }
}


// ====================================================
// RESET CAPTURE STATE FOR A NEW HANDHELD TEST
// Call only while BLE scanning is stopped.
// ====================================================

void resetRIDCaptureState() {

    if (targetMutex != nullptr) {
        xSemaphoreTake(
            targetMutex,
            portMAX_DELAY
        );
    }

    for (
        uint8_t i = 0;
        i < MAX_TARGETS;
        i++
    ) {
        targets[i] = RIDTarget();
    }

    if (targetMutex != nullptr) {
        xSemaphoreGive(
            targetMutex
        );
    }

    totalAdvertisements = 0;
    remoteIDPackets = 0;
    targetAllocationFailures = 0;
    lastDashboardPrintMs = 0;

    for (
        uint8_t i = 0;
        i < 16;
        i++
    ) {
        messageTypeCounts[i] = 0;
    }

    scanCompletePending = false;
    scanStopRequested = false;
    scanStartedMs = 0;
    scanWindowCount = 0;

    if (
        pBLEScan != nullptr
        &&
        !pBLEScan->isScanning()
    ) {
        pBLEScan->clearResults();
    }
}


// ====================================================
// BLE CALLBACK
// ====================================================

class RemoteIDCallbacks
    :
    public BLEAdvertisedDeviceCallbacks {

    void onResult(
        BLEAdvertisedDevice device
    ) override {

        totalAdvertisements++;


        // =================================================
        // ESP32-S3 / NimBLE RAW ADVERTISEMENT TRANSPORT
        //
        // ESP32 Arduino 3.3.11 NimBLE reuses advertiser
        // objects and parseAdvertisement() accumulates parsed
        // Service Data entries across duplicate reports.
        //
        // Scanner parsing is therefore disabled for the active
        // S3 harness. getPayload() now represents only the
        // current passive advertisement because setPayload()
        // replaces the previous raw payload when append=false.
        //
        // Walk the current BLE AD structures directly and pass
        // only validated 0xFFFA / 27-byte RID Service Data into
        // the accepted RID decoder path below.
        // =================================================

        const uint8_t *payload =
            device.getPayload();

        size_t payloadLength =
            device.getPayloadLength();


        if (
            payload != nullptr
            &&
            payloadLength > 0
        ) {

            size_t offset = 0;


            while (
                offset < payloadLength
            ) {

                uint8_t adLength =
                    payload[offset];


                // A zero-length field terminates the AD data.
                if (
                    adLength == 0
                ) {

                    break;
                }


                // adLength counts every byte after the length
                // byte: one AD type byte plus the AD data.
                size_t completeFieldLength =
                    static_cast<size_t>(
                        adLength
                    )
                    +
                    1;


                // Reject a malformed/truncated AD field before
                // accessing anything beyond the current buffer.
                if (
                    completeFieldLength
                    >
                    (
                        payloadLength
                        -
                        offset
                    )
                ) {

                    break;
                }


                // Every non-zero AD structure contains a type.
                uint8_t adType =
                    payload[
                        offset + 1
                    ];


                size_t adDataLength =
                    static_cast<size_t>(
                        adLength
                    )
                    -
                    1;


                const uint8_t *adData =
                    payload
                    +
                    offset
                    +
                    2;


                // Bluetooth AD Type 0x16:
                // Service Data - 16-bit UUID.
                if (
                    adType == 0x16
                    &&
                    adDataLength >= 2
                ) {

                    // Bluetooth 16-bit UUID is little-endian
                    // inside the Service Data AD structure.
                    uint16_t serviceUUID =
                        static_cast<uint16_t>(
                            adData[0]
                        )
                        |
                        (
                            static_cast<uint16_t>(
                                adData[1]
                            )
                            << 8
                        );


                    size_t serviceDataLength =
                        adDataLength
                        -
                        2;


                    const uint8_t *serviceDataBytes =
                        adData
                        +
                        2;


                    bool isRemoteIDUUID =
                        serviceUUID
                        ==
                        0xFFFA;


                    bool hasExpectedRemoteIDLength =
                        serviceDataLength
                        ==
                        27;


                    bool hasOpenDroneIDCode =

                        serviceDataLength >= 1

                        &&

                        serviceDataBytes[0]
                        ==
                        0x0D;


                    if (
                        isRemoteIDUUID
                        &&
                        hasOpenDroneIDCode
                        &&
                        hasExpectedRemoteIDLength
                    ) {

                        // Preserve binary RID bytes exactly,
                        // including any embedded zero values.
                        String serviceData(
                            reinterpret_cast<const char *>(
                                serviceDataBytes
                            ),
                            serviceDataLength
                        );


                        remoteIDPackets++;


                        String sourceAddress =
                            device
                            .getAddress()
                            .toString()
                            .c_str();


                        if (targetMutex != nullptr) {
                            xSemaphoreTake(
                                targetMutex,
                                portMAX_DELAY
                            );
                        }


                        int targetIndex =
                            findOrCreateTarget(
                                sourceAddress
                            );


                        if (
                            targetIndex < 0
                        ) {

                            if (targetMutex != nullptr) {
                                xSemaphoreGive(
                                    targetMutex
                                );
                            }

                            offset +=
                                completeFieldLength;

                            continue;
                        }


                        RIDTarget &target =
                            targets[
                                targetIndex
                            ];


                        target.packetCount++;

                        target.lastSeenMs =
                            millis();


                        addRSSISample(
                            target,
                            device.getRSSI()
                        );


                        uint8_t odidHeader =
                            getByte(
                                serviceData,
                                2
                            );


                        uint8_t messageType =
                            (
                                odidHeader >> 4
                            )
                            &
                            0x0F;


                        uint8_t protocolVersion =
                            odidHeader
                            &
                            0x0F;


                        target.protocolVersion =
                            protocolVersion;


                        messageTypeCounts[
                            messageType
                        ]++;


                        if (
                            messageType == 0
                        ) {

                            updateBasicID(
                                target,
                                serviceData
                            );
                        }


                        else if (
                            messageType == 1
                        ) {

                            updateLocation(
                                target,
                                serviceData
                            );
                        }


                        else if (
                            messageType == 4
                        ) {

                            updateSystem(
                                target,
                                serviceData
                            );
                        }


                        else if (
                            messageType == 5
                        ) {

                            updateOperatorID(
                                target,
                                serviceData
                            );
                        }


                        if (targetMutex != nullptr) {
                            xSemaphoreGive(
                                targetMutex
                            );
                        }
                    }
                }


                offset +=
                    completeFieldLength;
            }
        }


        // Periodic dashboard output is intentionally NOT
        // performed inside this BLE callback.
        //
        // serviceTargetDashboard() owns diagnostic output from
        // the main loop so USB Serial can never hold up RF work.
    }
};

void ridScanCompleteCallback(
    BLEScanResults results
) {
    (void)results;

    // Do not perform display/serial-heavy work inside
    // the BLE completion callback. A completed BLE window is
    // serviced from loop() and automatically rolled over unless
    // the operator requested STOP/HOME.
    scanCompletePending = true;
}


bool startRIDScanWindow(
    bool newOperatorSession
) {

    if (pBLEScan == nullptr) {
        Serial.println(
            "[ERROR] BLE scanner is not initialized."
        );

        return false;
    }

    if (pBLEScan->isScanning()) {
        Serial.println(
            "[INFO] BLE scan window is already running."
        );

        return false;
    }

    scanCompletePending = false;

    if (newOperatorSession) {
        scanStopRequested = false;
        scanStartedMs = millis();
        scanWindowCount = 0;
    }

    bool started =
        pBLEScan->start(
            SCAN_WINDOW_SECONDS,
            ridScanCompleteCallback,
            false
        );

    if (started) {
        scanSessionActive = true;
        scanWindowCount++;

        if (newOperatorSession) {
            Serial.println(
                "[SCAN] Operator-owned continuous RID session started."
            );
        }
        else {
            Serial.print(
                "[SCAN] Continuous RID window rollover -> #"
            );
            Serial.println(scanWindowCount);
        }
    }
    else {
        if (newOperatorSession) {
            scanSessionActive = false;
        }

        Serial.println(
            "[ERROR] Asynchronous RID scan window failed to start."
        );
    }

    return started;
}


bool startRIDScanAsync() {
    return startRIDScanWindow(true);
}


bool continueRIDScanAsync() {

    if (
        pBLEScan == nullptr ||
        !scanSessionActive ||
        scanStopRequested
    ) {
        return false;
    }

    // A completed window can leave result objects allocated inside
    // the BLE scanner. They are no longer needed because decoded RID
    // state already lives in targets[]. Clearing them also prevents
    // accumulation across a long-running handheld session.
    if (!pBLEScan->isScanning()) {
        pBLEScan->clearResults();
    }

    return startRIDScanWindow(false);
}


bool requestStopRIDScan() {

    if (
        pBLEScan == nullptr ||
        !scanSessionActive
    ) {
        return false;
    }

    scanStopRequested = true;

    bool stopped = true;

    if (pBLEScan->isScanning()) {
        stopped = pBLEScan->stop();
    }

    if (stopped) {
        // Bluedroid does not need to fire the completion
        // callback for our firmware to finish the session.
        scanCompletePending = true;
    }

    return stopped;
}


void finalizeRIDScan() {

    if (!scanSessionActive) {
        scanCompletePending = false;
        return;
    }

    scanSessionActive = false;
    scanCompletePending = false;

    Serial.println();
    Serial.println(
        "*** FINAL MULTI-TARGET SNAPSHOT ***"
    );

    printAllTargetDetails();

    Serial.println();

    if (scanStopRequested) {
        Serial.println(
            "=== RID CAPTURE STOPPED BY USER ==="
        );
    }
    else {
        // Defensive fallback only. Normal BLE window completion no
        // longer finalizes an operator session in Stage 4B-B2C.
        Serial.println(
            "=== RID CAPTURE ENDED BY CONTROLLER ==="
        );
    }

    Serial.print(
        "Capture elapsed ms: "
    );

    Serial.println(
        millis() - scanStartedMs
    );

    Serial.print(
        "Continuous BLE scan windows started: "
    );

    Serial.println(
        scanWindowCount
    );

    Serial.print(
        "Total BLE advertisement callbacks: "
    );

    Serial.println(
        totalAdvertisements
    );

    Serial.print(
        "Validated Remote ID packets: "
    );

    Serial.println(
        remoteIDPackets
    );

    Serial.print(
        "Targets Tracked: "
    );

    Serial.println(
        getTrackedTargetCount()
    );

    Serial.print(
        "Target Allocation Failures: "
    );

    Serial.println(
        targetAllocationFailures
    );

    Serial.println();
    Serial.println(
        "Remote ID message types observed:"
    );

    for (
        int type = 0;
        type < 16;
        type++
    ) {

        if (
            messageTypeCounts[type] > 0
        ) {

            Serial.print(
                "Type "
            );

            Serial.print(
                type
            );

            Serial.print(
                " ("
            );

            Serial.print(
                messageTypeName(type)
            );

            Serial.print(
                "): "
            );

            Serial.println(
                messageTypeCounts[type]
            );
        }
    }

    Serial.println();
    Serial.println(
        "=== END RID CAPTURE ==="
    );

    pBLEScan->clearResults();

    scanStopRequested = false;
}


void serviceRIDScanState() {

    if (!scanSessionActive) {
        return;
    }

    bool underlyingWindowEnded =
        scanCompletePending ||
        (
            pBLEScan != nullptr &&
            !pBLEScan->isScanning()
        );

    if (!underlyingWindowEnded) {
        return;
    }

    // STOP/HOME owns finalization. It must never be converted into
    // an automatic rollover.
    if (scanStopRequested) {
        finalizeRIDScan();
        return;
    }

    // Some BLE backends may report the completion callback a fraction
    // before isScanning() flips false. Keep the pending flag set and
    // let the next loop iteration perform the rollover safely.
    if (
        pBLEScan != nullptr &&
        pBLEScan->isScanning()
    ) {
        return;
    }

    scanCompletePending = false;

    if (!continueRIDScanAsync()) {
        // Do not silently pretend the receiver is still listening.
        // Preserve captured target data for inspection, but mark the
        // session inactive so the UI accurately reflects the failure.
        scanSessionActive = false;

        Serial.println(
            "[ERROR] Continuous RID rollover failed; scan session stopped."
        );
    }
}


// ============================================================
// STAGE 4B-B2B RESCAN CONTROL
// Privacy mode and selectedTarget are UI state and are deliberately
// preserved. Target/counter data starts fresh.
// ============================================================

bool restartRIDScanFromUI() {
    if (pBLEScan == nullptr) {
        Serial.println("[UI ERROR] BLE scanner is not initialized.");
        return false;
    }

    if (scanSessionActive) {
        bool stopped = true;

        if (pBLEScan->isScanning()) {
            stopped = pBLEScan->stop();
        }

        if (!stopped) {
            Serial.println("[UI ERROR] RESCAN could not stop active scan.");
            return false;
        }

        // RESCAN is not STOP/HOME: retire the old session without printing
        // its final report, then let the BLE host settle briefly.
        scanSessionActive = false;
        scanCompletePending = false;
        scanStopRequested = false;

        delay(50);
        scanCompletePending = false;
    }

    resetRIDCaptureState();

    bool started = startRIDScanAsync();

    if (started) {
        Serial.println("[SCAN] RESCAN started fresh RID capture.");
    }

    return started;
}

bool captureUITargetSnapshot(
    uint8_t index,
    UITargetSnapshot &snapshot
) {
    snapshot = UITargetSnapshot();

    if (index >= MAX_TARGETS) {
        return false;
    }

    if (targetMutex != nullptr) {
        xSemaphoreTake(targetMutex, portMAX_DELAY);
    }

    RIDTarget &target = targets[index];
    snapshot.allocated = target.allocated;

    if (target.allocated) {
        uint32_t age = millis() - target.lastSeenMs;

        if (age <= TARGET_ACTIVE_MS) {
            snapshot.stateCode = 1;
        }
        else if (age <= TARGET_STALE_MS) {
            snapshot.stateCode = 2;
        }
        else {
            snapshot.stateCode = 3;
        }

        snapshot.uaType = target.uaType;
        snapshot.flightStatus = target.flightStatus;
        snapshot.protocolVersion = target.protocolVersion;

        snapshot.hasPosition = target.hasPosition;
        snapshot.latitude = target.latitude;
        snapshot.longitude = target.longitude;

        snapshot.directionValid = target.directionValid;
        snapshot.direction = target.direction;

        snapshot.horizontalSpeedValid = target.horizontalSpeedValid;
        snapshot.horizontalSpeed = target.horizontalSpeed;

        snapshot.verticalSpeedValid = target.verticalSpeedValid;
        snapshot.verticalSpeed = target.verticalSpeed;

        snapshot.barometricAltitudeValid = target.barometricAltitudeValid;
        snapshot.barometricAltitude = target.barometricAltitude;

        snapshot.geodeticAltitudeValid = target.geodeticAltitudeValid;
        snapshot.geodeticAltitude = target.geodeticAltitude;

        snapshot.heightValid = target.heightValid;
        snapshot.height = target.height;
        snapshot.heightFromGround = target.heightFromGround;

        snapshot.currentRSSI = target.currentRSSI;
        snapshot.filteredRSSI = getFilteredRSSI(target);

        snapshot.packetCount = target.packetCount;
        snapshot.basicIDCount = target.basicIDCount;
        snapshot.locationCount = target.locationCount;
        snapshot.systemCount = target.systemCount;
        snapshot.operatorIDCount = target.operatorIDCount;
        snapshot.lastSeenMs = target.lastSeenMs;

        const char *idText = nullptr;

        if (target.hasSerialID) {
            idText = target.serialID.c_str();
        }
        else if (target.hasRegistrationID) {
            idText = target.registrationID.c_str();
        }
        else if (target.hasOperatorID) {
            idText = target.operatorID.c_str();
        }

        if (idText != nullptr) {
            snprintf(
                snapshot.displayID,
                sizeof(snapshot.displayID),
                "%s",
                idText
            );
            snapshot.displayIDValid = true;
        }
    }

    if (targetMutex != nullptr) {
        xSemaphoreGive(targetMutex);
    }

    return snapshot.allocated;
}

int medianValues(
    int values[],
    int count
) {
    int temp[TAP_SAMPLES];

    for (int i = 0; i < count; i++) {
        temp[i] = values[i];
    }

    for (int i = 0; i < count - 1; i++) {
        for (int j = i + 1; j < count; j++) {
            if (temp[j] < temp[i]) {
                int swapValue = temp[i];
                temp[i] = temp[j];
                temp[j] = swapValue;
            }
        }
    }

    return temp[count / 2];
}

bool mapTouchToScreen(
    double rawTouchX,
    double rawTouchY,
    double &screenX,
    double &screenY
) {
    double denominator =
        h31 * rawTouchX
        + h32 * rawTouchY
        + 1.0;

    if (fabs(denominator) < 0.000001) {
        return false;
    }

    screenX =
        (
            h11 * rawTouchX
            + h12 * rawTouchY
            + h13
        ) / denominator;

    screenY =
        (
            h21 * rawTouchX
            + h22 * rawTouchY
            + h23
        ) / denominator;

    return true;
}

void waitForRelease() {
    unsigned long startTime = millis();

    while (
        ts.touched()
        && millis() - startTime < 2500
    ) {
        delay(10);
    }

    delay(80);
}

bool captureFilteredTap(
    int &medianX,
    int &medianY,
    int &medianZ
) {
    if (!ts.tirqTouched() || !ts.touched()) {
        return false;
    }

    int sampleX[TAP_SAMPLES];
    int sampleY[TAP_SAMPLES];
    int sampleZ[TAP_SAMPLES];
    int validSamples = 0;
    unsigned long captureStart = millis();

    while (
        ts.touched()
        && validSamples < TAP_SAMPLES
        && millis() - captureStart < 400
    ) {
        TS_Point p = ts.getPoint();

        if (p.z >= PRESSURE_MIN) {
            sampleX[validSamples] = p.x;
            sampleY[validSamples] = p.y;
            sampleZ[validSamples] = p.z;
            validSamples++;
        }

        delay(10);
    }

    waitForRelease();

    if (validSamples < 3) {
        return false;
    }

    medianX = medianValues(sampleX, validSamples);
    medianY = medianValues(sampleY, validSamples);
    medianZ = medianValues(sampleZ, validSamples);

    return true;
}

void logButtonEvent(
    ButtonID button,
    double x,
    double y
) {
    tapCounter++;

    Serial.print("[UI] tap=");
    Serial.print(tapCounter);
    Serial.print(" button=");
    Serial.print(static_cast<int>(button));
    Serial.print(" x=");
    Serial.print(x, 1);
    Serial.print(" y=");

void processButton(
    ButtonID button
) {
    bool redraw = true;

    switch (button) {
        case BTN_BEGIN:
            selectedTarget = 0;
            frozenSnapshotCaptured = false;
            outputMode = MODE_PRIVATE;
            currentScreen = SCREEN_PRIVACY;
            Serial.println("ACTION: BEGIN TEST");
            break;

        case BTN_PRIVATE:
            outputMode = MODE_PRIVATE;
            Serial.println("ACTION: PRIVATE");
            break;

        case BTN_OPEN:
            outputMode = MODE_OPEN;
            Serial.println("ACTION: OPEN");
            break;

        case BTN_START_SCAN:
            if (scanSessionActive) {
                Serial.println("[UI] Scan already active.");
                break;
            }

            resetRIDCaptureState();
            frozenSnapshotCaptured = false;

            if (startRIDScanAsync()) {
                scanGeneration++;
                selectedTarget = 0;
                currentScreen = SCREEN_LIVE;
                Serial.println("ACTION: START SCAN");
            }
            else {
                Serial.println("[UI ERROR] START SCAN failed.");
            }
            break;

        case BTN_BACK_HOME:
            currentScreen = SCREEN_HOME;
            Serial.println("ACTION: BACK HOME");
            break;

        case BTN_MODE_TOGGLE:
            outputMode =
                outputMode == MODE_PRIVATE
                ? MODE_OPEN
                : MODE_PRIVATE;
            Serial.println("ACTION: MODE TOGGLE");
            break;

        case BTN_TARGET_1:
            selectedTarget = 0;
            Serial.println("ACTION: SELECT T01");
            break;

        case BTN_TARGET_2:
            selectedTarget = 1;
            Serial.println("ACTION: SELECT T02");
            break;

        case BTN_TARGET_3:
            selectedTarget = 2;
            Serial.println("ACTION: SELECT T03");
            break;

        case BTN_TARGET_4:
            selectedTarget = 3;
            Serial.println("ACTION: SELECT T04");
            break;

        case BTN_STOP:
            if (scanSessionActive) {
                requestStopRIDScan();
            }
            currentScreen = SCREEN_HOME;
            Serial.println("ACTION: STOP / HOME");
            break;

        case BTN_SNAPSHOT:
            frozenSnapshotTarget = selectedTarget;
            frozenSnapshotMode = outputMode;
            frozenSnapshotCapturedMs = millis();
            frozenSnapshotCaptured =
                captureUITargetSnapshot(
                    selectedTarget,
                    frozenSnapshot
                );
            currentScreen = SCREEN_SNAPSHOT;
            Serial.print("ACTION: SNAPSHOT T0");
            Serial.print(selectedTarget + 1);
            Serial.println(
                frozenSnapshotCaptured
                ? " FROZEN"
                : " EMPTY"
            );
            break;

        case BTN_RESCAN:
            // Preserve selectedTarget and outputMode across the restart.
            frozenSnapshotCaptured = false;

            if (restartRIDScanFromUI()) {
                scanGeneration++;
                Serial.println("ACTION: RESCAN");
            }
            else {
                Serial.println("[UI ERROR] RESCAN failed.");
            }
            break;

        case BTN_BACK_LIVE:
            currentScreen = SCREEN_LIVE;
            Serial.println("ACTION: BACK TO LIVE");
            break;

        default:
            redraw = false;
            Serial.println("ACTION: NONE");
            break;
    }

    if (redraw) {
        drawCurrentScreen();
        lastLiveUIRefreshMs = millis();
    }
}

void serviceTouchUI() {
    int rawX = 0;
    int rawY = 0;
    int rawZ = 0;

    if (!captureFilteredTap(rawX, rawY, rawZ)) {
        return;
    }

    double screenX = 0.0;
    double screenY = 0.0;

    if (
        !mapTouchToScreen(
            rawX,
            rawY,
            screenX,
            screenY
        )
    ) {
        Serial.println("ERROR: TOUCH MAP");
        return;
    }

    ButtonID button =
        hitTestCurrentScreen(
            screenX,
            screenY
        );

    logButtonEvent(
        button,
        screenX,
        screenY
    );

    processButton(button);
}

void serviceLiveUIRefresh() {
    if (currentScreen != SCREEN_LIVE) {
        return;
    }

    if (
        millis() - lastLiveUIRefreshMs
        < LIVE_UI_REFRESH_MS
    ) {
        return;
    }

    drawLiveScreen();
    lastLiveUIRefreshMs = millis();
}


// ====================================================

void setup() {

    Serial.begin(115200);
    delay(STARTUP_DELAY_MS);

    targetMutex =
        xSemaphoreCreateMutex();

    if (targetMutex == nullptr) {
        Serial.println(
            "[FATAL] target mutex allocation failed."
        );
    }

    // Validated CYD1 touch/display initialization.
    touchSPI.begin(
        XPT2046_CLK,
        XPT2046_MISO,
        XPT2046_MOSI,
        XPT2046_CS
    );
    ts.begin(touchSPI);
    ts.setRotation(1);

    tft.init();
    tft.setRotation(1);

    outputMode = MODE_PRIVATE;
    currentScreen = SCREEN_HOME;

    // BLE is initialized at boot, but scanning is operator-started.
    BLEDevice::init("");
    pBLEScan = BLEDevice::getScan();
    pBLEScan->setActiveScan(false);
    pBLEScan->setInterval(100);
    pBLEScan->setWindow(99);
    pBLEScan->setAdvertisedDeviceCallbacks(
        &ridCallbacks,
        true
    );

    Serial.println();
    Serial.println(
        "========================================"
    );
    Serial.println(
        " CYD1 RID HANDHELD - STAGE 4B-B2C"
    );
    Serial.println(
        " UI + RID + SNAPSHOT + CONTINUOUS SCAN"
    );
    Serial.println(
        "========================================"
    );
    Serial.println(
        "Boot state: HOME / BLE IDLE / PRIVATE"
    );
    Serial.println(
        "Touch START SCAN; operator STOP/HOME ends capture."
    );

    drawCurrentScreen();
}


// ====================================================
// LOOP
// ====================================================

void loop() {

    processSerialCommands();
    serviceRIDScanState();
    serviceTouchUI();
    serviceLiveUIRefresh();

    delay(15);
}

// ============================================================
// RADAR GEOGRAPHIC ENGINE - D4A
//
// Geographic calculations are based ONLY on:
//   - a valid receiver latitude/longitude reference
//   - a valid target latitude/longitude position
//
// RSSI is never used to infer range or bearing.
//
// The first RADAR implementation is NORTH-UP. True bearing is
// derived mathematically from the two geographic coordinates,
// so a physical compass is not required for north-up plotting.
// ============================================================

constexpr double WS_GEO_PI =
    3.14159265358979323846;

constexpr double WS_EARTH_RADIUS_METERS =
    6371008.8;


// Receiver reference intentionally starts invalid.
//
// Reference source is explicit so the UI and geographic engine
// never confuse a fixed bench location with future live GNSS.
//
// RADAR must remain unavailable until:
//   - the receiver reference is valid
//   - the reference source is known
//   - a target has a fresh valid geographic position

enum WSReceiverReferenceSource {

    WS_REF_NONE = 0,

    WS_REF_MANUAL_FIXED = 1,

    WS_REF_GNSS = 2
};


struct WSReceiverReference {

    bool valid = false;

    WSReceiverReferenceSource source =
        WS_REF_NONE;

    double latitude = 0.0;

    double longitude = 0.0;
};


struct WSGeographicSolution {

    bool valid = false;

    double distanceMeters = 0.0;

    double bearingTrueDegrees = 0.0;
};


// ------------------------------------------------------------
// D4D-B RADAR TARGET SOLUTION
//
// Fixed-size result derived only from:
//   - validated receiver reference coordinates
//   - a fresh validated target geographic position
//
// RSSI is never used for distance or bearing.
// ------------------------------------------------------------

    bool valid = false;

    int16_t x = 0;

// ============================================================
// D4D-C NORTH-UP RADAR COORDINATE PROJECTION
// ============================================================

bool wsProjectRadarTarget(
    const WSRadarTargetSolution &solution,
    int16_t centerX,
    int16_t centerY,
    int16_t radiusPixels,
    double rangeMeters,
    WSRadarPlotPoint &point
) {

    point =
        WSRadarPlotPoint();


    if (
        !solution.valid
        ||
        radiusPixels <= 0
        ||
        rangeMeters <= 0.0
        ||
        solution.distanceMeters < 0.0
        ||
        solution.bearingTrueDegrees < 0.0
        ||
        solution.bearingTrueDegrees >= 360.0
    ) {

        return false;
    }


    double plottedDistance =
        solution.distanceMeters;


    if (
        plottedDistance
        > rangeMeters
    ) {

        plottedDistance =
            rangeMeters;

        point.offScale =
            true;
    }


    const double radialFraction =
        plottedDistance
        /
        rangeMeters;


    const double radialPixels =
        radialFraction
        *
        static_cast<double>(
            radiusPixels
        );


    const double bearingRadians =
        wsDegreesToRadians(
            solution.bearingTrueDegrees
        );


    // North-up screen transform:
    //
    // 0 deg   -> up
    // 90 deg  -> right
    // 180 deg -> down
    // 270 deg -> left

    const double xOffset =
        sin(
            bearingRadians
        )
        *
        radialPixels;

    const double yOffset =
        -cos(
            bearingRadians
        )
        *
        radialPixels;


    point.x =
        static_cast<int16_t>(
            lround(
                static_cast<double>(
                    centerX
                )
                +
                xOffset
            )
        );


    point.y =
        static_cast<int16_t>(
            lround(
                static_cast<double>(
                    centerY
                )
                +
                yOffset
            )
        );


    point.plottedDistanceMeters =
        plottedDistance;

    point.valid =
        true;

    return true;
}


bool wsCoordinateValid(
    double latitude,
    double longitude
) {

    return (
        latitude >= -90.0
        &&
        latitude <= 90.0
        &&
        longitude >= -180.0
        &&
        longitude <= 180.0
    );
}


void wsClearReceiverReference() {

    wsReceiverReference.valid =
        false;

    wsReceiverReference.source =
        WS_REF_NONE;

    wsReceiverReference.latitude =
        0.0;

    wsReceiverReference.longitude =
        0.0;
}


bool wsSetManualFixedReceiverReference(
    double latitude,
    double longitude
) {

    if (
        !wsCoordinateValid(
            latitude,
            longitude
        )
    ) {

        wsClearReceiverReference();

        return false;
    }


    wsReceiverReference.latitude =
        latitude;

    wsReceiverReference.longitude =
        longitude;

    wsReceiverReference.source =
        WS_REF_MANUAL_FIXED;

    wsReceiverReference.valid =
        true;

    return true;
}


uint8_t wsReceiverReferenceStatusCode() {

    if (
        !wsReceiverReference.valid
        ||
        wsReceiverReference.source
            == WS_REF_NONE
    ) {

        return 0;
    }


    if (
        wsReceiverReference.source
            == WS_REF_MANUAL_FIXED
    ) {

        return 1;
    }


    if (
        wsReceiverReference.source
            == WS_REF_GNSS
    ) {

        return 2;
    }


    return 0;
}


double wsDegreesToRadians(
    double degrees
) {

    return (
        degrees
        * WS_GEO_PI
        / 180.0
    );
}


double wsRadiansToDegrees(
    double radians
) {

    return (
        radians
        * 180.0
        / WS_GEO_PI
    );
}


bool wsSolveGeographic(
    double receiverLatitude,
    double receiverLongitude,
    double targetLatitude,
    double targetLongitude,
    WSGeographicSolution &solution
) {

    solution.valid = false;

    solution.distanceMeters = 0.0;

    solution.bearingTrueDegrees = 0.0;


    if (
        !wsCoordinateValid(
            receiverLatitude,
            receiverLongitude
        )
        ||
        !wsCoordinateValid(
            targetLatitude,
            targetLongitude
        )
    ) {

        return false;
    }


    const double receiverLatRad =
        wsDegreesToRadians(
            receiverLatitude
        );

    const double targetLatRad =
        wsDegreesToRadians(
            targetLatitude
        );

    const double deltaLatRad =
        wsDegreesToRadians(
            targetLatitude
            - receiverLatitude
        );

    const double deltaLonRad =
        wsDegreesToRadians(
            targetLongitude
            - receiverLongitude
        );


    const double sinHalfDeltaLat =
        sin(
            deltaLatRad
            / 2.0
        );

    const double sinHalfDeltaLon =
        sin(
            deltaLonRad
            / 2.0
        );


    double haversineA =
        (
            sinHalfDeltaLat
            * sinHalfDeltaLat
        )
        +
        (
            cos(
                receiverLatRad
            )
            *
            cos(
                targetLatRad
            )
            *
            sinHalfDeltaLon
            *
            sinHalfDeltaLon
        );


    // Numerical protection near antipodal points.

    if (haversineA < 0.0) {

        haversineA = 0.0;
    }

    else if (haversineA > 1.0) {

        haversineA = 1.0;
    }


    const double centralAngle =
        2.0
        *
        atan2(
            sqrt(
                haversineA
            ),
            sqrt(
                1.0
                - haversineA
            )
        );


    solution.distanceMeters =
        WS_EARTH_RADIUS_METERS
        *
        centralAngle;


    const double bearingY =
        sin(
            deltaLonRad
        )
        *
        cos(
            targetLatRad
        );

    const double bearingX =
        (
            cos(
                receiverLatRad
            )
            *
            sin(
                targetLatRad
            )
        )
        -
        (
            sin(
                receiverLatRad
            )
            *
            cos(
                targetLatRad
            )
            *
            cos(
                deltaLonRad
            )
        );


    double bearingDegrees =
        wsRadiansToDegrees(
            atan2(
                bearingY,
                bearingX
            )
        );


    bearingDegrees =
        fmod(
            bearingDegrees
            + 360.0,
            360.0
        );


    solution.bearingTrueDegrees =
        bearingDegrees;

    solution.valid = true;

    return true;
}


// RADAR is intentionally unavailable at startup.
//
// Later this becomes data-driven using:
//   - valid receiver reference position
//   - fresh valid target geographic position
//   - any additional stability criteria we adopt.
//
// It must NEVER become ready from RSSI alone.

bool wsRadarReady = false;


// Geographic freshness is intentionally based on the timestamp
// of the most recent valid Location/Vector position update,
// NOT the timestamp of arbitrary RID traffic.
//
// Reuse the existing ACTIVE freshness interval so geographic
// readiness follows the same conservative 3-second freshness
// expectation already used elsewhere in the receiver.

const uint32_t WS_RADAR_POSITION_FRESH_MS =
    TARGET_ACTIVE_MS;


// ============================================================
// D4D-B PER-TARGET RADAR GEOGRAPHIC SOLUTION
// ============================================================

bool wsBuildRadarTargetSolution(
    uint8_t targetIndex,
    const WSUITargetSnapshot &snapshot,
    WSRadarTargetSolution &result
) {

    result =
        WSRadarTargetSolution();

    result.targetIndex =
        targetIndex;


    if (targetIndex >= MAX_TARGETS) {

        return false;
    }


    // Receiver reference must be explicitly valid and sourced.

    if (
        !wsReceiverReference.valid
        ||
        wsReceiverReference.source
            == WS_REF_NONE
        ||
        !wsCoordinateValid(
            wsReceiverReference.latitude,
            wsReceiverReference.longitude
        )
    ) {

        return false;
    }


    // Target must contain a real geographic position and a
    // position-specific freshness timestamp.

    if (
        !snapshot.allocated
        ||
        !snapshot.hasPosition
        ||
        snapshot.lastPositionSeenMs == 0
    ) {

        return false;
    }


    if (
        !wsCoordinateValid(
            snapshot.latitude,
            snapshot.longitude
        )
    ) {

        return false;
    }


    const uint32_t positionAge =
        millis()
        -
        snapshot.lastPositionSeenMs;


    if (
        positionAge
        > WS_RADAR_POSITION_FRESH_MS
    ) {

        return false;
    }


    // Reuse the already-validated D4A great-circle solver.

    WSGeographicSolution geographicSolution;


    if (
        !wsSolveGeographic(
            wsReceiverReference.latitude,
            wsReceiverReference.longitude,
            snapshot.latitude,
            snapshot.longitude,
            geographicSolution
        )
        ||
        !geographicSolution.valid
    ) {

        return false;
    }


    result.distanceMeters =
        geographicSolution.distanceMeters;

    result.bearingTrueDegrees =
        geographicSolution.bearingTrueDegrees;


    // Carry the existing OpenDroneID height semantics forward.
    //
    // heightFromGround == true  -> AGL
    // heightFromGround == false -> ATO

    result.heightValid =
        snapshot.heightValid;

    result.heightMeters =
        snapshot.height;

    result.heightFromGround =
        snapshot.heightFromGround;


    result.valid =
        true;

    return true;
}


bool wsEvaluateRadarReady() {

    // RADAR belongs to an operator-owned active capture session.

    if (
        !wsUiSessionStarted
        ||
        !scanSessionActive
    ) {

        return false;
    }


    // A reference must exist, identify its source, and contain
    // valid geographic coordinates.

    if (
        !wsReceiverReference.valid
        ||
        wsReceiverReference.source
            == WS_REF_NONE
        ||
        !wsCoordinateValid(
            wsReceiverReference.latitude,
            wsReceiverReference.longitude
        )
    ) {

        return false;
    }


    const uint32_t now =
        millis();

    bool freshGeographicTarget =
        false;


    if (targetMutex != nullptr) {

        xSemaphoreTake(
            targetMutex,
            portMAX_DELAY
        );
    }


    for (
        uint8_t i = 0;
        i < MAX_TARGETS;
        i++
    ) {

        const RIDTarget &target =
            targets[i];


        if (
            !target.allocated
            ||
            !target.hasPosition
            ||
            target.lastPositionSeenMs == 0
        ) {

            continue;
        }


        if (
            !wsCoordinateValid(
                target.latitude,
                target.longitude
            )
        ) {

            continue;
        }


        const uint32_t positionAge =
            now
            -
            target.lastPositionSeenMs;


        if (
            positionAge
            <= WS_RADAR_POSITION_FRESH_MS
        ) {

            freshGeographicTarget =
                true;

            break;
        }
    }


    if (targetMutex != nullptr) {

        xSemaphoreGive(
            targetMutex
        );
    }


    return freshGeographicTarget;
}


void serviceWaveshareRadarReadiness() {

    const bool readyNow =
        wsEvaluateRadarReady();


    if (
        readyNow
        == wsRadarReady
    ) {

        return;
    }


    wsRadarReady =
        readyNow;


    if (wsRadarReady) {

        Serial.println(
            "[RADAR] READY: valid receiver reference + fresh geographic target."
        );
    }

    else {

        Serial.println(
            "[RADAR] ACQUIRING: geographic prerequisites unavailable."
        );
    }


    // Update only the tab/header when readiness changes.
    // Do not disturb the live target panel or BLE receiver.

    if (
        wsCurrentScreen
        == WS_SCREEN_LIVE
    ) {

        wsDrawLiveTabs();
    }
}

// ============================================================
// FIXED RID TARGET SNAPSHOT
// ============================================================

bool wsCaptureTargetSnapshot(
    uint8_t index,
    WSUITargetSnapshot &snapshot
) {

    snapshot =
        WSUITargetSnapshot();


    if (index >= MAX_TARGETS) {

        return false;
    }


    if (targetMutex != nullptr) {

        xSemaphoreTake(
            targetMutex,
            portMAX_DELAY
        );
    }


    RIDTarget &target =
        targets[index];


    snapshot.allocated =
        target.allocated;


    if (target.allocated) {

        uint32_t age =
            millis()
            - target.lastSeenMs;


        if (age <= TARGET_ACTIVE_MS) {

            snapshot.stateCode = 1;
        }

        else if (age <= TARGET_STALE_MS) {

            snapshot.stateCode = 2;
        }

        else {

            snapshot.stateCode = 3;
        }


        snapshot.uaType =
            target.uaType;

        snapshot.flightStatus =
            target.flightStatus;

        snapshot.protocolVersion =
            target.protocolVersion;


        snapshot.hasPosition =
            target.hasPosition;

        snapshot.latitude =
            target.latitude;

        snapshot.longitude =
            target.longitude;


        snapshot.directionValid =
            target.directionValid;

        snapshot.direction =
            target.direction;


        snapshot.horizontalSpeedValid =
            target.horizontalSpeedValid;

        snapshot.horizontalSpeed =
            target.horizontalSpeed;


        snapshot.verticalSpeedValid =
            target.verticalSpeedValid;

        snapshot.verticalSpeed =
            target.verticalSpeed;


        snapshot.barometricAltitudeValid =
            target.barometricAltitudeValid;

        snapshot.barometricAltitude =
            target.barometricAltitude;


        snapshot.geodeticAltitudeValid =
            target.geodeticAltitudeValid;

        snapshot.geodeticAltitude =
            target.geodeticAltitude;


        snapshot.heightValid =
            target.heightValid;

        snapshot.height =
            target.height;

        snapshot.heightFromGround =
            target.heightFromGround;


        snapshot.currentRSSI =
            target.currentRSSI;

        snapshot.filteredRSSI =
            getFilteredRSSI(
                target
            );


        snapshot.packetCount =
            target.packetCount;

        snapshot.basicIDCount =
            target.basicIDCount;

        snapshot.locationCount =
            target.locationCount;

        snapshot.systemCount =
            target.systemCount;

        snapshot.operatorIDCount =
            target.operatorIDCount;

        snapshot.lastSeenMs =
            target.lastSeenMs;

        snapshot.lastPositionSeenMs =
            target.lastPositionSeenMs;


        const char *idText =
            nullptr;


        if (target.hasSerialID) {

            idText =
                target.serialID.c_str();
        }

        else if (target.hasRegistrationID) {

            idText =
                target.registrationID.c_str();
        }

        else if (target.hasOperatorID) {

            idText =
                target.operatorID.c_str();
        }


        if (idText != nullptr) {

            snprintf(
                snapshot.displayID,
                sizeof(snapshot.displayID),
                "%s",
                idText
            );

            snapshot.displayIDValid =
                true;
        }
    }


    if (targetMutex != nullptr) {

        xSemaphoreGive(
            targetMutex
        );
    }


    return snapshot.allocated;
}


// ============================================================
// HIT TEST HELPER
// ============================================================

bool wsPointInside(
    int16_t x,
    int16_t y,
    const WSUIButton &button
) {

    // Small invisible expansion helps finger use without
    // creating excessive overlap between adjacent controls.

    const int16_t padding = 4;

    return (
        x >= button.x - padding
        && x <= button.x + button.w + padding
        && y >= button.y - padding
        && y <= button.y + button.h + padding
    );
}


// ============================================================
// GENERIC BUTTON DRAW
// ============================================================

void wsDrawButton(
    const WSUIButton &button,
    bool selected,
    uint16_t borderColor
) {

    uint16_t fillColor =
        selected
        ? CYAN
        : BLACK;

    uint16_t textColor =
        selected
        ? BLACK
        : WHITE;


    wsGfx->fillRoundRect(
        button.x,
        button.y,
        button.w,
        button.h,
        8,
        fillColor
    );


    wsGfx->drawRoundRect(
        button.x,
        button.y,
        button.w,
        button.h,
        8,
        selected
        ? CYAN
        : borderColor
    );


    wsGfx->setTextSize(
        button.textSize
    );

    wsGfx->setTextColor(
        textColor
    );


    int16_t textWidth =
        strlen(button.label)
        * 6
        * button.textSize;

    int16_t textHeight =
        8
        * button.textSize;


    wsGfx->setCursor(
        button.x
        + (button.w - textWidth) / 2,

        button.y
        + (button.h - textHeight) / 2
    );


    wsGfx->print(
        button.label
    );
}

void wsDrawLiveTabs() {

    // D4D-D2B:
    // Keep the original large touch hitboxes, but visually
    // compact the tabs while RADAR is the active live view.

    const int16_t wsLiveTabDrawHeight =
        (
            wsCurrentLiveView
            == WS_VIEW_RADAR
        )
        ? 38

// BLE capture and touch handling remain independent.

    if (
        wsCurrentScreen
            != WS_SCREEN_LIVE

        || wsCurrentLiveView
            != WS_VIEW_TARGETS
    ) {

        return;
    }


    uint32_t now =
        millis();


    if (
        now
        -
        wsLastLiveTargetRefreshMs
        <

        now;

// ============================================================
// RADAR VIEW
// ============================================================

void wsDrawRadarView() {

    // ========================================================
    // D4D-D2A RADAR-FIRST LAYOUT
    //
    // LEFT:
    //   Large north-up geographic scope.
    //
    // RIGHT:
    //   Per-target geographic details.
    //
    // MODE and STOP remain at the lower-right using the
    // existing large finger-friendly buttons.
    //
    // SNAPSHOT and RESCAN belong to TARGETS view and are not
    // rendered here.
    // ========================================================


    // --------------------------------------------------------
    // LEFT RADAR PANEL
    // --------------------------------------------------------

    wsGfx->fillRect(
        10,
        96,
        225,
        214,
        BLACK
    );


    wsGfx->drawRoundRect(
        10,
        96,
        225,
        214,
        8,
        GREEN
    );


    // --------------------------------------------------------
    // RIGHT TARGET-DATA PANEL
    //
    // Stops above MODE / STOP so periodic RADAR refreshes do
    // not erase those controls.
    // --------------------------------------------------------

    wsGfx->fillRect(
        245,
        96,
        225,
        154,
        BLACK
    );


    wsGfx->drawRoundRect(
        245,
        96,
        225,
        154,
        8,
        GREEN
    );


    const int16_t cx =
        WS_RADAR_CENTER_X;

    const int16_t cy =
        WS_RADAR_CENTER_Y;

    const int16_t radius =
        WS_RADAR_RADIUS_PX;


    // --------------------------------------------------------
    // Geographic scope
    // --------------------------------------------------------

    // --------------------------------------------------------
    // Subdued quadrant reference lines
    //
    // Drawn BEFORE the range rings so they remain background
    // structure rather than competing with target/range data.
    // --------------------------------------------------------

    wsGfx->drawLine(
        cx - radius,
        cy,
        cx + radius,
        cy,
        WS_COLOR_DARK_GREY
    );


    wsGfx->drawLine(
        cx,
        cy - radius,
        cx,
        cy + radius,
        WS_COLOR_DARK_GREY
    );


    // Proportional range rings:
    //
    //   inner  = 1/3 displayed range
    //   middle = 2/3 displayed range
    //   outer  = full displayed range
    //
    // No distance labels are placed on the scope.
    // RANGE defines the outer-ring scale.

    wsGfx->drawCircle(
        cx,
        cy,
        radius / 3,
        WS_COLOR_GREY
    );


    wsGfx->drawCircle(
        cx,
        cy,
        (radius * 2) / 3,
        WS_COLOR_GREY
    );


    wsGfx->drawCircle(
        cx,
        cy,
        radius,
        WHITE
    );


    // Receiver-reference marker.

    wsGfx->fillCircle(
        cx,
        cy,
        4,
        WHITE
    );


    // --------------------------------------------------------
    // Cardinal labels
    // --------------------------------------------------------

    wsGfx->setTextSize(1);
    wsGfx->setTextColor(GREEN);


    wsGfx->setCursor(
        cx - 3,
        cy - radius - 10
    );

    wsGfx->print(
        "N"
    );


    wsGfx->setCursor(
        cx + radius + 6,
        cy - 14
    );

    wsGfx->print(
        "E"
    );


    wsGfx->setCursor(
        cx - 3,
        cy + radius + 3
    );

    wsGfx->print(
        "S"
    );


    wsGfx->setCursor(
        cx - radius - 12,
        cy - 14
    );

    wsGfx->print(
        "W"
    );


    // --------------------------------------------------------
    // Right-panel fixed status fields.
    // --------------------------------------------------------

    wsGfx->setTextColor(
        WHITE
    );


    wsGfx->setCursor(
        255,
        103
    );

    wsGfx->print(
        "RANGE:"
    );

    wsGfx->print(
        wsRadarDisplayRangeMeters,
        0
    );

    wsGfx->print(
        "m"
    );


    wsGfx->setCursor(
        350,
        103
    );

    wsGfx->print(
        "REF:"
    );


    if (
        wsReceiverReference.valid
        &&
        wsReceiverReference.source
            == WS_REF_MANUAL_FIXED
    ) {

        wsGfx->print(
            "FIXED"
        );
    }

    else if (
        wsReceiverReference.valid
        &&
        wsReceiverReference.source
            == WS_REF_GNSS
    ) {

        wsGfx->print(
            "GNSS"
        );
    }

    else {

        wsGfx->print(
            "NONE"
        );
    }


    wsGfx->drawLine(
        255,
        117,
        460,
        117,
        WS_COLOR_DARK_GREY
    );


    // --------------------------------------------------------
    // Build, project and display each target.
    // --------------------------------------------------------

    uint8_t plottedTargetCount =
        0;


    for (
        uint8_t i = 0;
        i < MAX_TARGETS;
        i++
    ) {

        const int16_t rowY =
            125
            +
            static_cast<int16_t>(i) * 27;


        WSUITargetSnapshot snapshot;


        bool snapshotValid =
            wsCaptureTargetSnapshot(
                i,
                snapshot
            );


        // ----------------------------------------------------
        // Production visibility rule:
        //
        // A slot that has never been allocated does not
        // represent a real detected target and must remain
        // completely invisible in RADAR view.
        //
        // Allocated targets continue below, where a temporary
        // lack of fresh geographic data is shown truthfully as
        // NO FRESH GEO.
        // ----------------------------------------------------

        if (
            !snapshotValid
            ||
            !snapshot.allocated
        ) {

            continue;
        }


        // ----------------------------------------------------
        // Production lifecycle visibility rule:
        //
        // ACTIVE:
        //   Plot when geographic data is fresh.
        //
        // STALE:
        //   Keep the real target identity visible, but do not
        //   plot stale geographic coordinates. The existing
        //   NO FRESH GEO path handles this truthfully.
        //
        // LOST:
        //   Remove the target from RADAR presentation entirely.
        //   The backend slot remains untouched and may still be
        //   reacquired/reused according to existing lifecycle
        //   logic.
        // ----------------------------------------------------

        if (snapshot.stateCode == 3) {

            continue;
        }


        WSRadarTargetSolution solution;


        bool solutionValid =
            wsBuildRadarTargetSolution(
                i,
                snapshot,
                solution
            );


        if (!solutionValid) {

            wsGfx->setTextColor(
                WS_COLOR_GREY
            );

            wsGfx->setCursor(
                255,
                rowY
            );

            wsGfx->print(
                "T0"
            );

            wsGfx->print(
                i + 1
            );

            wsGfx->print(
                "  NO FRESH GEO"
            );


            wsGfx->setCursor(
                255,
                rowY + 11
            );

            wsGfx->print(
                "DIST/BRG/HGT --"
            );

            continue;
        }


        WSRadarPlotPoint point;


        if (
            !wsProjectRadarTarget(
                solution,
                cx,
                cy,
                radius,
                wsRadarDisplayRangeMeters,
                point
            )
        ) {

            wsGfx->setTextColor(
                WS_COLOR_GREY
            );

            wsGfx->setCursor(
                255,
                rowY
            );

            wsGfx->print(
                "T0"
            );

            wsGfx->print(
                i + 1
            );

            wsGfx->print(
                "  PLOT ERROR"
            );

            continue;
        }


        plottedTargetCount++;


        const uint16_t targetColor =
            point.offScale
            ? YELLOW
            : CYAN;


        // ----------------------------------------------------
        // Scope marker
        // ----------------------------------------------------

        wsGfx->fillCircle(
            point.x,
            point.y,
            4,
            targetColor
        );


        wsGfx->setTextColor(
            targetColor
        );


        int16_t labelX =
            point.x
            + 6;

        int16_t labelY =
            point.y
            - 4;


        if (
            labelX
            >
            210
        ) {

            labelX =
                point.x
                -
                24;
        }


        if (
            labelY
            <
            112
        ) {

            labelY =
                point.y
                +
                7;
        }


        if (
            labelY
            >
            292
        ) {

            labelY =
                point.y
                -
                12;
        }


        wsGfx->setCursor(
            labelX,
            labelY
        );

        wsGfx->print(
            "T0"
        );

        wsGfx->print(
            i + 1
        );


        // ----------------------------------------------------
        // Right-side target information
        //
        // Line 1:
        //   Target ID / actual distance / true bearing
        //
        // Line 2:
        //   truthful AGL or ATO height in feet when available
        // ----------------------------------------------------

        const long distanceMetersRounded =
            lround(
                solution.distanceMeters
            );


        int bearingRounded =
            static_cast<int>(
                lround(
                    solution.bearingTrueDegrees
                )
            );


        if (
            bearingRounded
            >= 360
        ) {

            bearingRounded -=
                360;
        }


        wsGfx->setCursor(
            255,
            rowY
        );

        wsGfx->print(
            "T0"
        );

        wsGfx->print(
            i + 1
        );

        wsGfx->print(
            " "
        );

        wsGfx->print(
            distanceMetersRounded
        );

        wsGfx->print(
            "m B"
        );


        if (
            bearingRounded
            < 100
        ) {

            wsGfx->print(
                "0"
            );
        }


        if (
            bearingRounded
            < 10
        ) {

            wsGfx->print(
                "0"
            );
        }


        wsGfx->print(
            bearingRounded
        );


        if (
            point.offScale
        ) {

            wsGfx->print(
                " OFF"
            );
        }


        wsGfx->setCursor(
            255,
            rowY + 11
        );


        if (
            solution.heightValid
        ) {

            const long heightFeet =
                lround(
                    static_cast<double>(
                        solution.heightMeters
                    )
                    *
                    3.28084
                );


            wsGfx->print(
                solution.heightFromGround
                ? "AGL "
                : "ATO "
            );

            wsGfx->print(
                heightFeet
            );

            wsGfx->print(
                "ft"
            );
        }

        else {

            wsGfx->setTextColor(
                WS_COLOR_GREY
            );

            wsGfx->print(
                "HGT UNKNOWN"
            );
        }
    }


    // --------------------------------------------------------
    // Plot count / orientation
    // --------------------------------------------------------

    wsGfx->setTextColor(
        WHITE
    );

    wsGfx->setCursor(
        255,
        229
    );

    wsGfx->print(
        "PLOTS:"
    );

    wsGfx->print(
        plottedTargetCount
    );


    wsGfx->setCursor(
        390,
        229
    );

    wsGfx->print(
        "N-UP"
    );


    wsLastRadarRefreshMs =
        millis();
}

// ============================================================
// SNAPSHOT SCREEN
//
// Renders ONLY the frozen fixed-size snapshot captured by D1.
// The live BLE receiver continues running underneath this view.
// ============================================================

const char *wsSnapshotStateName(
    const WSUITargetSnapshot &snapshot
) {

    switch (snapshot.stateCode) {

        case 1:
            return "ACTIVE";

        case 2:
            return "STALE";

        case 3:
            return "LOST";

        default:
            return "EMPTY";
    }
}


void wsDrawSnapshotScreen() {

    wsGfx->fillScreen(

// ============================================================
// WAVESHARE BUTTON PROCESSING
//
// Stage 1B-B owns real privacy mode and capture START / STOP.
// RESCAN owns a real fresh backend restart.
// SNAPSHOT freezes a mutex-protected target copy and opens
// a dedicated frozen-data screen while BLE remains active.
// ============================================================

void wsProcessButton(
    WSButtonID button
) {

    bool redraw = true;


    switch (button) {

        case WS_BTN_BEGIN:

            outputMode = MODE_PRIVATE;
            wsUiPrivateSelected = true;
            wsUiSelectedTarget = 0;
            wsUiSessionStarted = false;
            wsFrozenSnapshotCaptured = false;
            wsRadarReady = false;
            wsCurrentLiveView = WS_VIEW_TARGETS;

            wsCurrentScreen =
                WS_SCREEN_PRIVACY;

            Serial.println(
                "[UI] BEGIN -> OUTPUT MODE / backend PRIVATE / scanner IDLE."
            );

            break;


        case WS_BTN_PRIVATE:

            outputMode = MODE_PRIVATE;
            wsUiPrivateSelected = true;

            Serial.println(
                "[UI] Output mode -> PRIVATE."
            );

            break;


        case WS_BTN_OPEN:

            outputMode = MODE_OPEN;
            wsUiPrivateSelected = false;

            Serial.println(
                "[UI] Output mode -> OPEN."
            );

            break;


        case WS_BTN_START:

            if (scanSessionActive) {

                Serial.println(
                    "[UI] START ignored: RID session already active."
                );

                redraw = false;
                break;
            }


            resetRIDCaptureState();

            wsUiSessionStarted = false;
            wsRadarReady = false;

            wsCurrentLiveView =
                WS_VIEW_TARGETS;


            if (startRIDScanAsync()) {

                wsUiSessionStarted = true;

                wsCurrentScreen =
                    WS_SCREEN_LIVE;

                Serial.println(
                    "[UI] START SCAN -> TARGETS / backend ACTIVE."
                );

                Serial.println(
                    "[UI] RADAR state = ACQUIRING."
                );
            }

            else {

                Serial.println(
                    "[UI ERROR] START SCAN failed; remaining on OUTPUT MODE."
                );
            }

            break;


        case WS_BTN_BACK:

            wsCurrentScreen =
                WS_SCREEN_HOME;

            Serial.println(
                "[UI-SHELL] BACK -> HOME"
            );

            break;


        case WS_BTN_BACK_LIVE:

            wsCurrentScreen =
                WS_SCREEN_LIVE;

            Serial.println(
                "[UI] SNAPSHOT -> BACK TO LIVE."
            );

            break;


        case WS_BTN_TAB_TARGETS:

            wsCurrentLiveView =
                WS_VIEW_TARGETS;

            Serial.println(
                "[UI-SHELL] VIEW -> TARGETS"
            );

            break;


        case WS_BTN_TAB_RADAR:

            if (!wsRadarReady) {

                redraw = false;

                Serial.println(
                    "[UI-SHELL] RADAR unavailable: ACQUIRING geographic prerequisites."
                );

                break;
            }


            wsCurrentLiveView =
                WS_VIEW_RADAR;

            Serial.println(
                "[UI-SHELL] VIEW -> RADAR"
            );

            break;


        case WS_BTN_T01:

            wsUiSelectedTarget = 0;

            Serial.println(
                "[UI-SHELL] SELECT T01"
            );

            break;


        case WS_BTN_T02:

            wsUiSelectedTarget = 1;

            Serial.println(
                "[UI-SHELL] SELECT T02"
            );

            break;


        case WS_BTN_T03:

            wsUiSelectedTarget = 2;

            Serial.println(
                "[UI-SHELL] SELECT T03"
            );

            break;


        case WS_BTN_T04:

            wsUiSelectedTarget = 3;

            Serial.println(
                "[UI-SHELL] SELECT T04"
            );

            break;


        case WS_BTN_SNAPSHOT:

            if (!scanSessionActive) {

                Serial.println(
                    "[UI] SNAPSHOT ignored: no active RID session."
                );

                break;
            }


            wsFrozenSnapshotTarget =
                wsUiSelectedTarget;

            wsFrozenSnapshotMode =
                outputMode;

            wsFrozenSnapshotCapturedMs =
                millis();


            wsFrozenSnapshotCaptured =
                wsCaptureTargetSnapshot(
                    wsUiSelectedTarget,
                    wsFrozenSnapshot
                );


            Serial.print(
                "[UI] SNAPSHOT T0"
            );

            Serial.print(
                wsUiSelectedTarget + 1
            );

            Serial.print(
                wsFrozenSnapshotCaptured
                ? " FROZEN / "
                : " EMPTY / "
            );

            Serial.println(
                wsFrozenSnapshotMode
                    == MODE_PRIVATE
                ? "PRIVATE"
                : "OPEN"
            );


            wsCurrentScreen =
                WS_SCREEN_SNAPSHOT;

            break;


        case WS_BTN_RESCAN:

            if (!scanSessionActive) {

                redraw = false;

                Serial.println(
                    "[UI] RESCAN ignored: no active RID session."
                );

                break;
            }


            // A RESCAN creates a new capture generation.
            // Do not carry a frozen target from the prior
            // generation into the fresh session.
            wsFrozenSnapshotCaptured = false;


            wsRadarReady = false;


            if (restartRIDScanFromUI()) {

                wsUiSessionStarted = true;

                Serial.println(
                    "[UI] RESCAN complete / fresh RID capture ACTIVE."
                );

                Serial.println(
                    "[UI] RADAR state = ACQUIRING."
                );
            }

            else {

                // Mirror the backend's actual state.
                // A failed RESCAN may occur before the existing
                // scan session has been successfully retired.
                wsUiSessionStarted =
                    scanSessionActive;

                Serial.println(
                    "[UI ERROR] RESCAN failed."
                );
            }

            break;


        case WS_BTN_MODE:

            outputMode =
                outputMode == MODE_PRIVATE
                ? MODE_OPEN
                : MODE_PRIVATE;

            wsUiPrivateSelected =
                outputMode == MODE_PRIVATE;

            Serial.println(
                outputMode == MODE_PRIVATE
                ? "[UI] Output mode -> PRIVATE."
                : "[UI] Output mode -> OPEN."
            );

            break;


        case WS_BTN_STOP:

            if (scanSessionActive) {

                if (requestStopRIDScan()) {

                    Serial.println(
                        "[UI] STOP / HOME requested."
                    );
                }

                else {

                    Serial.println(
                        "[UI ERROR] STOP request failed."
                    );
                }
            }

            else {

                Serial.println(
                    "[UI] STOP / HOME: receiver already idle."
                );
            }


            wsUiSessionStarted = false;
            wsRadarReady = false;

            wsCurrentLiveView =
                WS_VIEW_TARGETS;

            wsCurrentScreen =
                WS_SCREEN_HOME;

            break;


        default:

            redraw = false;
            break;
    }


    if (redraw) {

        wsDrawCurrentScreen();
    }
}


// ============================================================
// CAPACITIVE TOUCH SERVICE
//
// Action requires:
//   DOWN inside a control
//   UP inside the SAME control
//
// This gives us basic protection against accidental drags
// and hand movement.
// ============================================================

void serviceWaveshareTouchUI() {

    if (
        millis() - wsLastTouchPollMs
        < WS_TOUCH_POLL_MS
    ) {

        return;
    }


    wsLastTouchPollMs =
        millis();


    int16_t rawX[1];
    int16_t rawY[1];


    uint8_t touched =
        wsTouch.getPoint(
            rawX,
            rawY,
            1
        );


    if (touched) {

        // Proven Rotation-1 transform:
        //
        // screen X = raw Y
        // screen Y = 319 - raw X

        int16_t screenX =
            constrain(
                rawY[0],
                0,
                wsGfx->width() - 1
            );


        int16_t screenY =
            constrain(
                (WS_LCD_HOR_RES - 1)
                - rawX[0],
                0,
                wsGfx->height() - 1
            );


        wsTouchLastX =
            screenX;

        wsTouchLastY =
            screenY;


        if (!wsTouchWasDown) {

            wsTouchWasDown = true;


            wsTouchDownButton =
                wsHitTest(
                    screenX,
                    screenY
                );


            Serial.printf(
                "[TOUCH-DOWN] x=%d y=%d button=%s\n",
                screenX,
                screenY,
                wsButtonName(
                    wsTouchDownButton
                )
            );
        }


        return;
    }


    if (!wsTouchWasDown) {

        return;
    }


    wsTouchWasDown =
        false;


    WSButtonID releaseButton =
        wsHitTest(
            wsTouchLastX,
            wsTouchLastY
        );


    Serial.printf(
        "[TOUCH-UP] x=%d y=%d button=%s\n",
        wsTouchLastX,
        wsTouchLastY,
        wsButtonName(
            releaseButton
        )
    );


    if (
        wsTouchDownButton
            != WS_BTN_NONE

        && releaseButton
            == wsTouchDownButton
    ) {

        wsProcessButton(
            releaseButton
        );
    }

    else {

        Serial.println(
            "[UI-SHELL] TOUCH CANCELLED"
        );
    }


    wsTouchDownButton =
        WS_BTN_NONE;
}

// ============================================================
// ESP32-S3 BACKEND + DISPLAY MIGRATION HARNESS
// ============================================================

void setup() {

    Serial.begin(115200);

    // ESP32-S3 native USB HWCDC:
    // Prevent unconsumed Serial TX from stalling the receiver
    // when no serial monitor is actively reading output.
    Serial.setTxTimeoutMs(0);

    delay(STARTUP_DELAY_MS);


    targetMutex =
        xSemaphoreCreateMutex();

    if (targetMutex == nullptr) {
        Serial.println(
            "[FATAL] target mutex allocation failed."
        );

        while (true) {
            delay(1000);
        }
    }

    outputMode = MODE_PRIVATE;


    // ------------------------------------------------
    // Waveshare display bring-up
    // Proven hardware path from Touch_Bringup.
    // Touch is intentionally not enabled yet.
    // ------------------------------------------------

    Wire.begin(
        WS_I2C_SDA,
        WS_I2C_SCL
    );

    wsTCA.begin();
    wsTCA.pinMode1(
        1,
        OUTPUT
    );

    wsLcdReset();

    if (!wsGfx->begin()) {

        Serial.println(
            "[FATAL] Waveshare gfx->begin() failed."
        );

        while (true) {
            delay(1000);
        }
    }

    pinMode(
        WS_GFX_BL,
        OUTPUT
    );

    digitalWrite(
        WS_GFX_BL,
        HIGH
    );

    // First physical-orientation candidate:
    // landscape, rotation 1.
    wsGfx->setRotation(1);

    wsDrawCurrentScreen();

    Serial.printf(
        "[DISPLAY] Rotation 1 / width=%d / height=%d\n",
        wsGfx->width(),
        wsGfx->height()
    );

    if (!wsTouch.begin(
            Wire,
            FT6X36_SLAVE_ADDRESS
        )) {

        Serial.println(
            "[FATAL] FT6X36 touch controller not found."
        );

        while (true) {
            delay(1000);
        }
    }

    Serial.println(
        "[TOUCH] FT6X36 detected / TARGETS-RADAR UI shell active."
    );


    BLEDevice::init("");

    pBLEScan =
        BLEDevice::getScan();

    pBLEScan->setActiveScan(false);
    pBLEScan->setInterval(100);
    pBLEScan->setWindow(99);

    pBLEScan->setAdvertisedDeviceCallbacks(
        &ridCallbacks,
        true,
        false
    );

    resetRIDCaptureState();

    Serial.println();
    Serial.println(
        "========================================"
    );
    Serial.println(
        " ESP32-S3 RID BACKEND PARITY HARNESS"
    );
    Serial.println(
        " ACCEPTED CYD1 RID ENGINE / NO UI"
    );
    Serial.println(
        "========================================"
    );
    Serial.println(
        "Boot state: HOME / BLE READY / PRIVATE / IDLE"
    );
    Serial.println(
        "Serial commands P/O/S remain available."
    );
    Serial.println();

    // Stage 1B-A:
    // BLE is initialized and ready, but the receiver remains IDLE
    // until the operator explicitly starts a capture session.
    //
    // Do NOT auto-start RID scanning from setup().
    scanSessionActive = false;
    scanCompletePending = false;
    scanStopRequested = false;

    Serial.println(
        "[SCAN] Receiver idle - awaiting operator START."
    );
}
